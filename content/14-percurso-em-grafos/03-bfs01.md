## Ideia

Arestas de peso 0 vão para a frente do deque e as de peso 1 para o fim, mantendo a ordem de distância.

## Código

```cpp
vector<int> bfs01(const vector<vector<pair<int, int>>>& g, int s) {
    vector<int> d(g.size(), INT_MAX);
    deque<int> q{s}; d[s] = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop_front();
        for (auto [v, w] : g[u]) if (d[u] + w < d[v]) {
            d[v] = d[u] + w;
            w ? q.push_back(v) : q.push_front(v);
        }
    }
    return d;
}
```
