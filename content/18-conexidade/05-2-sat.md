## Ideia

Cada cláusula `(a ∨ b)` equivale às implicações `¬a → b` e `¬b → a`. Crie um vértice por literal (`2i` para `x_i`, `2i+1` para `¬x_i`). A fórmula é satisfatível se e só se nenhuma variável está na mesma SCC que sua negação. Para montar uma atribuição, como a numeração de Tarjan é topológica reversa, escolha `x_i = verdadeiro` quando `comp[x_i] < comp[¬x_i]`.

## Código

```cpp
struct DoisSat {
    int n;
    vector<vector<int>> g;
    DoisSat(int n) : n(n), g(2 * n) {}
    // literal: variável i com valor val (true = x_i, false = ¬x_i)
    static int lit(int i, bool val) { return 2 * i + (val ? 0 : 1); }
    // (x_a == va) OU (x_b == vb)
    void clausula(int a, bool va, int b, bool vb) {
        g[lit(a, va) ^ 1].push_back(lit(b, vb));
        g[lit(b, vb) ^ 1].push_back(lit(a, va));
    }
    void forca(int a, bool va) { clausula(a, va, a, va); }  // x_a == va

    // devolve false se insatisfatível; senão preenche valor[i]
    bool resolve(vector<bool>& valor) {
        int N = 2 * n, timer = 0, nc = 0;
        vector<int> tin(N, -1), low(N, 0), it(N, 0), pilha, comp(N, -1);
        vector<char> naPilha(N, false);
        for (int s = 0; s < N; s++) {
            if (tin[s] != -1) continue;
            vector<int> chamada = {s};
            tin[s] = low[s] = timer++; pilha.push_back(s); naPilha[s] = true;
            while (!chamada.empty()) {
                int u = chamada.back();
                if (it[u] < (int)g[u].size()) {
                    int v = g[u][it[u]++];
                    if (tin[v] == -1) {
                        tin[v] = low[v] = timer++; pilha.push_back(v); naPilha[v] = true;
                        chamada.push_back(v);
                    } else if (naPilha[v]) low[u] = min(low[u], tin[v]);
                } else {
                    if (low[u] == tin[u]) {
                        while (true) {
                            int v = pilha.back(); pilha.pop_back();
                            naPilha[v] = false; comp[v] = nc;
                            if (v == u) break;
                        }
                        nc++;
                    }
                    chamada.pop_back();
                    if (!chamada.empty()) low[chamada.back()] = min(low[chamada.back()], low[u]);
                }
            }
        }
        valor.assign(n, false);
        for (int i = 0; i < n; i++) {
            if (comp[2 * i] == comp[2 * i + 1]) return false;
            valor[i] = comp[2 * i] < comp[2 * i + 1];
        }
        return true;
    }
};
```

## Complexidade e cuidados

- Variáveis numeradas `0..n-1`. Para `x -> y` use a cláusula `(¬x ∨ y)`; para `x XOR y` use `(x ∨ y)` e `(¬x ∨ ¬y)`.
- Cada cláusula adiciona 2 arestas; 3 ou mais literais por cláusula torna o problema NP-completo.
- A atribuição devolvida é uma das possíveis; não é a lexicograficamente menor.
- Cláusula com a mesma variável duas vezes (`forca`) fixa o valor da variável.
